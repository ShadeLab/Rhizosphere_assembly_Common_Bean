# (1) calculate Shannon index for genes at each stage
awk '
NR==1 {
    print "Sample\tShannon_Index" > "gene_shannon.txt";
    for (i=2; i<=NF; i++) samples[i]=$i
} 
NR>1 {
    for (i=2; i<=NF; i++) abundance[i] += $i
    for (i=2; i<=NF; i++) count[i, NR] = $i
} 
END {
    for (i=2; i<=NF; i++) {
        H = 0;
        for (j=2; j<=NR; j++) {
            if (count[i, j] > 0) {
                p = count[i, j] / abundance[i];
                H -= p * log(p);  # Shannon index
            }
        }
        print samples[i] "\t" H > "gene_shannon.txt";
    }
}' gene.TPM
